package main

import (
	"reflect"
	"testing"
)

func TestXMLImportsVerifierCases(t *testing.T) {
	for _, test := range []struct {
		name string
		body string
	}{
		{
			name: "gpx",
			body: `<gpx><trk><name>Harbour café</name><trkseg><trkpt lat="-33.8" lon="151.2"/><trkpt lat="-33.81" lon="151.21"/></trkseg></trk></gpx>`,
		},
		{
			name: "tcx",
			body: `<TrainingCenterDatabase><Courses><Course><Name>Harbour café</Name><Track><Trackpoint><Position><LatitudeDegrees>-33.8</LatitudeDegrees><LongitudeDegrees>151.2</LongitudeDegrees></Position></Trackpoint><Trackpoint><Position><LatitudeDegrees>-33.81</LatitudeDegrees><LongitudeDegrees>151.21</LongitudeDegrees></Position></Trackpoint></Track></Course></Courses></TrainingCenterDatabase>`,
		},
		{
			name: "kml",
			body: `<kml><Document><name>Harbour café</name><Placemark><LineString><coordinates>151.2,-33.8 151.21,-33.81</coordinates></LineString></Placemark></Document></kml>`,
		},
	} {
		t.Run(test.name, func(t *testing.T) {
			name := "harbour-" + test.name + ".xml"
			original, err := loadTrack(importFile(t, name, []byte(test.body)))
			if err != nil {
				t.Fatal(err)
			}
			if len(original.Points) != 2 {
				t.Fatal("independent input lost its two track points")
			}
			if original.Points[0].Lat != -33.8 || original.Points[0].Lon != 151.2 {
				t.Fatal("independent input changed its coordinates")
			}
			for _, prefix := range []string{"\xef\xbb\xbf", "\xef\xbb\xbf<?xml version=\"1.0\"?>\r\n<!-- Harbour -->\n"} {
				track, err := loadTrack(importFile(t, name, []byte(prefix+test.body)))
				if err != nil || !reflect.DeepEqual(track, original) {
					t.Fatal("independent BOM import changed or failed", err)
				}
			}
			for _, prefix := range []string{"\n\xef\xbb\xbf", "\xef\xbb\xbf\xef\xbb\xbf", "\xef\xbb\xbfunexpected"} {
				if _, err := loadTrack(importFile(t, name, []byte(prefix+test.body))); err == nil {
					t.Fatal("independent import accepted invalid leading text")
				}
			}
		})
	}
}
