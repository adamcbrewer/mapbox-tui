package main

import "testing"

func TestHarnessImportBounds(t *testing.T) {
	for _, test := range []struct {
		name   string
		body   string
		points []Point
		bounds [4]float64
	}{
		{
			name:   "north-east",
			body:   `<gpx><trk><trkseg><trkpt lat="32" lon="44"/><trkpt lat="33" lon="46"/></trkseg></trk></gpx>`,
			points: []Point{{Lat: 32, Lon: 44}, {Lat: 33, Lon: 46}},
			bounds: [4]float64{44, 32, 46, 33},
		},
		{
			name:   "crosses-origin",
			body:   `<gpx><trk><trkseg><trkpt lat="-2" lon="-3"/><trkpt lat="4" lon="5"/></trkseg></trk></gpx>`,
			points: []Point{{Lat: -2, Lon: -3}, {Lat: 4, Lon: 5}},
			bounds: [4]float64{-3, -2, 5, 4},
		},
	} {
		t.Run(test.name, func(t *testing.T) {
			track, err := loadTrack(importFile(t, "journey.gpx", []byte(test.body)))
			if err != nil {
				t.Fatal(err)
			}
			if len(track.Points) != len(test.points) || len(track.Segments) != 1 || len(track.Waypoints) != 0 {
				t.Fatal("imported geometry changed")
			}
			for i, point := range track.Points {
				if point.Lat != test.points[i].Lat || point.Lon != test.points[i].Lon {
					t.Fatal("imported coordinates changed")
				}
			}
			if track.Bounds != test.bounds {
				t.Fatalf("import bounds include unrelated locations: got %v, want %v", track.Bounds, test.bounds)
			}
		})
	}
}
